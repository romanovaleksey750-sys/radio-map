# -*- coding: utf-8 -*- import folium import pandas as pd import math from branca.element import Template, MacroElement # --- Ваши координаты приёма --- RX_LAT, RX_LON = 56.851607, 36.687092 # --- Загрузка данных --- df = pd.read_csv("stations.csv", encoding="utf-8") df.columns = df.columns.str.strip() # --- Базовая карта (Esri — проверенно работает) --- m = folium.Map( location=[RX_LAT, RX_LON], zoom_start=8, tiles='https://server.arcgisonline.com/ArcGIS/rest/services/World_Street_Map/MapServer/tile/{z}/{y}/{x}', attr='Tiles © Esri — Source: Esri, i-cubed, USDA, USGS, AEX, GeoEye, Getmapping, Aerogrid, IGN, IGP, UPR-EGP, and the GIS User Community' ) # --- Приёмный пункт --- folium.Marker( [RX_LAT, RX_LON], popup="📻 Приёмный пункт
пос. 1 Мая, Конаковский р-н", icon=folium.Icon(color="green", icon="radio", prefix="fa") ).add_to(m) # --- Кольца расстояний --- for r in [25, 50, 100, 200, 300]: folium.Circle( [RX_LAT, RX_LON], radius=r * 1000, color="gray", fill=False, weight=1, opacity=0.3, tooltip=f"{r} км" ).add_to(m) # --- Функция рисования дуги --- def draw_arc(parent_group, p1, p2, color, weight, tooltip, css_class=None): lat1, lon1 = p1 lat2, lon2 = p2 mid_lat = (lat1 + lat2) / 2 mid_lon = (lon1 + lon2) / 2 dx = lon2 - lon1 dy = lat2 - lat1 length = math.hypot(dx, dy) if length == 0: return perp_lat = -dx / length perp_lon = dy / length bend = 0.15 ctrl_lat = mid_lat + perp_lat * length * bend ctrl_lon = mid_lon + perp_lon * length * bend points = [] for i in range(21): t = i / 20 lat = (1-t)**2 * lat1 + 2*(1-t)*t * ctrl_lat + t**2 * lat2 lon = (1-t)**2 * lon1 + 2*(1-t)*t * ctrl_lon + t**2 * lon2 points.append([lat, lon]) line = folium.PolyLine( points, color=color, weight=weight, opacity=0.7, tooltip=tooltip ) if css_class: line.options['className'] = css_class line.add_to(parent_group) # --- Цвет для линий --- def color_for(dist): if dist < 25: return "#2e7d32" if dist < 60: return "#1565c0" if dist < 150: return "#ef6c00" return "#c62828" # --- Цвет для иконок --- def icon_color_for(dist): if dist < 25: return "green" if dist < 60: return "blue" if dist < 150: return "orange" return "red" # --- Группируем по городам --- cities = df.groupby("city").agg({ "lat": "mean", "lon": "mean", "freq": "count", "dist_km": "mean" }).reset_index() cities.columns = ["city", "lat", "lon", "count", "avg_dist"] # --- Создаём FeatureGroup для каждого города --- city_groups = {} for _, city_row in cities.iterrows(): city_name = city_row["city"] fg = folium.FeatureGroup(name=city_name, show=True) city_lat = float(city_row["lat"]) city_lon = float(city_row["lon"]) avg_dist = float(city_row["avg_dist"]) line_color = color_for(avg_dist) draw_arc(fg, [RX_LAT, RX_LON], [city_lat, city_lon], color=line_color, weight=2, tooltip=f"{city_name} — {int(city_row['count'])} станций, ср. {avg_dist:.0f} км", css_class="radio-arc") # HTML-таблица станций city_stations = df[df["city"] == city_name].sort_values("freq") rows_html = "" for _, s in city_stations.iterrows(): rows_html += ( f"" f"{s['freq']}" f"{s['station']}" f"{s['dist_km']:.0f} км" f"" ) popup_html = ( f"
" f"{city_name} " f"({int(city_row['count'])} станций)" f"" f"" f"" f"" f"" f"{rows_html}
МГцСтанцияРасст.
" ) # Иконка города — радиовышка (если не отобразится — замени на "signal") folium.Marker( [city_lat, city_lon], popup=folium.Popup(popup_html, max_width=400), tooltip=f"{city_name} — {int(city_row['count'])} станций", icon=folium.Icon(color=icon_color_for(avg_dist), icon="broadcast-tower", prefix="fa") ).add_to(fg) city_groups[city_name] = fg # --- Добавляем все группы на карту --- for fg in city_groups.values(): fg.add_to(m) # --- Панель управления слоями --- folium.LayerControl(collapsed=False).add_to(m) # --- Легенда --- legend = """ {% macro html(this, kwargs) %}
📻 Легенда
━ до 25 км
━ 25–60 км
━ 60–150 км
━ более 150 км
Клик по городу — список станций
Бегущий пунктир — направление сигнала
{% endmacro %} """ macro_legend = MacroElement() macro_legend._template = Template(legend) m.get_root().add_child(macro_legend) # --- CSS-анимация бегущего пунктира --- anim_css = """ {% macro html(this, kwargs) %} {% endmacro %} """ macro_anim = MacroElement() macro_anim._template = Template(anim_css) m.get_root().add_child(macro_anim) # --- Сохранение --- m.save("radio_map.html") print("Готово! Карта Esri, анимация к приёмнику.")
